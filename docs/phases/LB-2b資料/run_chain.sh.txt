run() {  # model label
  echo "=== $2 ===" ; date
  uv run python "$CHECK" "$1" || { echo "STOP: loaded models not exactly $1"; return 1; }

}

swap() {  # model
  echo "=== swap to $1 ===" ; date
  "$LMS" unload --all

}

}
