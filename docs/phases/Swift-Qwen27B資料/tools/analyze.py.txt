SIMPLIFIED = set(SIMPLIFIED_CHARS) | EXTRA_SIMPLIFIED

SKIP_MAINLAND_CHECK = {"賊", "整個人", "視野", "打造", "高端"}  # 台灣也常用，只計不算

def strings(obj):
    if isinstance(obj, str):
        yield obj
    elif isinstance(obj, dict):
        for v in obj.values():
            yield from strings(v)
    elif isinstance(obj, list):
        for v in obj:
            yield from strings(v)


def cjk_text(content):
    if content is None:
        return ""
    try:
        data = json.loads(content)
        parts = [s for s in strings(data) if re.search(r"[一-鿿]", s)]
        return "\n".join(parts)
    except (ValueError, TypeError):
        return content


def pct(values, p):
    if not values:
        return None
    s = sorted(values)
    k = max(0, min(len(s) - 1, int(round(p / 100 * len(s) + 0.5)) - 1))
    return s[k]


def main():
    out_path, *paths = sys.argv[1:]
    report = {}
    for path in paths:
        calls = [json.loads(line) for line in open(path, encoding="utf-8")]
        by_tag = defaultdict(list)
        for c in calls:
            by_tag[c["tag"]].append(c)
        for tag, cs in by_tag.items():
            secs = [c["seconds"] for c in cs if "seconds" in c]
            finish = Counter(c.get("finish", "error" if "error" in c else "?") for c in cs)
            errors = Counter((c.get("error") or "")[:60] for c in cs if "error" in c)
            json_ok = 0
            json_total = 0
            simp_hits = []
            main_hits = []
            with_cjk = 0
            tps = []
            for i, c in enumerate(cs):
                content = c.get("content")
                if c.get("mode"):
                    json_total += 1
                    try:
                        json.loads(content)
                        json_ok += 1
                    except (ValueError, TypeError):
                        pass
                text = cjk_text(content)
                if not text:
                    continue
                with_cjk += 1
                sc = sorted({ch for ch in text if ch in SIMPLIFIED})
                if sc:
                    idx = text.find(sc[0])
                    simp_hits.append({"i": i, "chars": "".join(sc), "context": text[max(0, idx - 15): idx + 15]})
                mh = sorted({w for w in MAINLAND if w in text})
                if mh:
                    w = mh[0]
                    idx = text.find(w)
                    main_hits.append({"i": i, "words": mh, "context": text[max(0, idx - 15): idx + 15]})
                st = c.get("stats") or {}
                if st.get("tokens_per_second"):
                    tps.append(st["tokens_per_second"])
            real_main = [h for h in main_hits if any(w not in SKIP_MAINLAND_CHECK for w in h["words"])]
            report[tag] = {
                "calls": len(cs),
                "p50": pct(secs, 50), "p90": pct(secs, 90), "max": max(secs) if secs else None,
                "mean": round(statistics.mean(secs), 2) if secs else None,
                "finish": dict(finish), "http_errors": dict(errors),
                "json_ok": json_ok, "json_total": json_total,
                "with_cjk": with_cjk,
                "simplified_calls": len(simp_hits), "simplified_examples": simp_hits[:30],
                "mainland_calls": len(real_main), "mainland_all": len(main_hits),
                "mainland_examples": main_hits[:40],
                "mainland_counter": dict(Counter(w for h in main_hits for w in h["words"])),
                "tps_median": statistics.median(tps) if tps else None,
            }
    with open(out_path, "w", encoding="utf-8", newline="\n") as f:
        json.dump(report, f, ensure_ascii=False, indent=2)
    for tag, r in report.items():
        print(tag, {k: v for k, v in r.items() if not k.endswith("examples")})


if __name__ == "__main__":
    main()
